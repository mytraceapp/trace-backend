javascriptfunction deduplicateResponse(proposedResponse, conversationHistory) {
  
  // 1. Extract the core semantic intent of the proposed response
  // What is TRACE actually saying/asking?
  
  // 2. Scan last 6 TRACE responses for semantic similarity
  // Not just exact matches — similar meaning counts
  // "are you safe?" and "are you okay right now?" are the same
  // "your feelings are valid" and "that's a heavy burden" are the same
  // "you don't have to go through this alone" and 
  // "you're not alone in it" are the same
  
  const recentAssistantMessages = conversationHistory
    .filter(m => m.role === 'assistant')
    .slice(-6)
    .map(m => (m.content || '').toLowerCase());
  
  // 3. Check for repeated PHRASES (not just questions)
  const SEMANTIC_CLUSTERS = [
    // Safety cluster — only say once per crisis
    ['are you safe', 'are you okay right now', 'safe right now',
     'are you in a safe place'],
    
    // 988 cluster — only say once ever
    ['988', 'crisis line', 'call or text'],
    
    // Not alone cluster — only say twice max
    ['not alone', "don't have to face", "don't have to go through"],
    
    // Feelings valid cluster — banned entirely  
    ['feelings are valid', 'okay to feel', 'feelings are important',
     'your feelings'],
    
    // Heavy burden cluster — only say once
    ['heavy burden', 'heavy weight', "that's a lot to carry",
     'heavy to carry'],
    
    // Share more cluster — only say once
    ['share more', 'share a bit more', 'tell me more',
     'can you share', 'share what'],
    
    // Here to listen cluster — banned after first use
    ['here to listen', 'here for you', 'here to support'],
    
    // Same thing acknowledgment — if user says its not helping
    // TRACE must change approach entirely
    ['same thing', "not helping", "isn't helping", "doesn't help"]
  ];
  
  const violations = [];
  
  for (const cluster of SEMANTIC_CLUSTERS) {
    // Check if proposed response contains this cluster
    const proposedHasCluster = cluster.some(phrase => 
      proposedResponse.toLowerCase().includes(phrase)
    );
    
    if (!proposedHasCluster) continue;
    
    // Check if recent history already used this cluster
    const historyHasCluster = recentAssistantMessages.some(msg =>
      cluster.some(phrase => msg.includes(phrase))
    );
    
    if (historyHasCluster) {
      violations.push(cluster[0]); // Log what was repeated
    }
  }
  
  // 4. If violations found — don't just flag, REPLACE
  if (violations.length > 0) {
    console.log(`[DEDUP] Repeated clusters: ${violations.join(', ')} — forcing new approach`);
    return {
      isDuplicate: true,
      violations,
      // Signal to regenerate or use fallback
    };
  }
  
  return { isDuplicate: false, violations: [] };
}
